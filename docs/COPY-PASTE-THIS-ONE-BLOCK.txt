    function renderSel(){
      var card=document.getElementById('as-selectedCard'),form=document.getElementById('as-selectionForm'),all=[];
      P.forEach(function(pl){pl.t.forEach(function(t){if(sel[t.id])all.push(t);});});
      if(!all.length){
        card.innerHTML='<p class="as-emptySelected">No songs selected yet. Choose from the playlists above.</p>';
        if(form)form.style.display='none';
        return;
      }
      card.innerHTML=all.map(function(t){return'<div class="as-selectedTrackItem">'+esc(t.n)+' — '+esc(t.a)+'</div>';}).join('');
      if(form)form.style.display='block';
    }
    document.getElementById('as-submitBtn').onclick=function(){
      var c=Object.keys(sel).length;
      if(!c){
        alert('Please select at least one song before submitting.');
        return;
      }
      var nameEl=document.getElementById('as-fullNameInput'),name=nameEl?nameEl.value.trim():'';
      if(!name){
        if(nameEl)nameEl.focus();
        alert('Please enter your full name before submitting.');
        return;
      }
      var emailEl=document.getElementById('as-emailInput'),email=emailEl?emailEl.value.trim():'';
      if(!email){
        if(emailEl)emailEl.focus();
        alert('Please enter your email before submitting.');
        return;
      }
      var labelEl=document.getElementById('as-selectionLabelInput'),label=labelEl?labelEl.value.trim():'';
      var ids=Object.keys(sel);
      var songsText=[];
      P.forEach(function(pl){pl.t.forEach(function(t){if(sel[t.id])songsText.push(t.n+' — '+t.a);});});
      var btn=document.getElementById('as-submitBtn');
      var orig=btn?btn.textContent:''; if(btn)btn.disabled=true; if(btn)btn.textContent='Submitting…';
      fetch((API_BASE||'').replace(/\/$/,'')+'/api/music-selection',{
        method:'POST',
        headers:{'Content-Type':'application/json'},
        body:JSON.stringify({name:name,email:email,label:label||undefined,songIds:ids,songsText:songsText})
      }).then(function(r){
        if(r.ok){
          if(typeof window!=='undefined' && window.location){
            window.location.href='https://aurorasonnet.com/repertoire-submissions-thank-you';
          }else{
            alert('Thank you! Your selections have been saved.');
            if(window.location&&window.location.reload)window.location.reload();
          }
          return;
        }else{
          r.json().catch(function(){return {};}).then(function(d){
            var msg=(d&&d.error)||('Something went wrong ('+r.status+').');
            alert(msg+' Please try again or email us directly.');
          });
        }
      }).catch(function(e){alert('Could not connect. Please check your connection and try again, or email us directly.');})
      .finally(function(){if(btn){btn.disabled=false;btn.textContent=orig;}});
    };
